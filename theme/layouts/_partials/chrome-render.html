{{/* @context {page} page @context {string} region navbar, footer, or sidebar @returns {bool} whether REGION renders on this page in the current chrome build mode (td.chrome); shared mode keeps each region on its donor page only. Guide: docsy.dev/content/en/docs/deployment/chrome.md */ -}} {{ $p := .page -}} {{ $render := true -}} {{ if eq ($p.Param "td.chrome") "shared" -}} {{ if eq .region "sidebar" -}} {{ $navRoot := cond (and (ne $p.Params.toc_root true) (eq $p.Site.Home.Type "docs")) $p.Site.Home $p.FirstSection -}} {{ $render = and $navRoot (eq $p.RelPermalink $navRoot.RelPermalink) -}} {{ else -}} {{/* The canonical render only, since the page's other renders carry a placeholder; tested by elimination over the alternatives, not with _partials/outputformat.html, which misreports pages that have an LLMS output. */ -}} {{ $canonical := $p.OutputFormats.Canonical -}} {{ $isCanonicalRender := and $canonical (not ($p.AlternativeOutputFormats.Get $canonical.Name)) -}} {{ $rootPage := partialCached "td/root-page.html" $p.Site $p.Site.Home.Permalink -}} {{ $render = and $isCanonicalRender ($p.Eq $rootPage) -}} {{ end -}} {{ end -}} {{ return $render -}}
