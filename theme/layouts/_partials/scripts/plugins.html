{{/* Emits each plugin registered in params.docsy.plugins. Registry contract, shape guards, and companion conventions: https://www.docsy.dev/project/implementation/script-loading/ */ -}} {{/* The theme declares its plugins under this key (theme/hugo.yaml), so a site's non-map value here, nil included, has already replaced them in Hugo's config merge: say so. */ -}} {{ $registry := dict -}} {{ $docsy := .Site.Params.docsy -}} {{ if not (reflect.IsMap $docsy) -}} {{ warnidf "docsy-plugins-config" "params.docsy must be a map of theme settings (got %v): the theme plugins are off. Docsy reserves this key; move your own value to another one." $docsy -}} {{ else if not (reflect.IsMap $docsy.plugins) -}} {{ warnidf "docsy-plugins-config" "params.docsy.plugins must be a map of plugin entries keyed by name (got %v): the theme plugins are off; write `plugins: {}` to keep them." $docsy.plugins -}} {{ else -}} {{ $registry = $docsy.plugins -}} {{ end -}} {{/* Normalize entries to a fixed lowercase key set (Hugo lowercases params keys, and `merge` yields plain maps without Hugo's case-insensitive access), then order by weight, then name. */ -}} {{ $entries := slice -}} {{ range $name, $value := $registry -}} {{ $entry := dict "enable" true "defer" false "pagegate" "" "weight" 0 "options" dict -}} {{ if reflect.IsMap $value -}} {{ range $k, $v := $value }}{{ $entry = merge $entry (dict (lower $k) $v) }}{{ end -}} {{ else if or (eq $value false) (eq (lower (printf "%v" $value)) "false") -}} {{ $entry = merge $entry (dict "enable" false) -}} {{ else -}} {{/* A scalar replaced the theme's entry at merge time; the intent is unrecoverable. */ -}} {{ warnidf "docsy-plugin-entry" "Plugin %q: a scalar value (%v) replaces the theme's entry and loses its defaults; write `%s: { enable: true }` instead. Skipping it." $name $value $name -}} {{ continue -}} {{ end -}} {{/* The key is the name; set last so an entry field can't redirect the asset. */ -}} {{ $entry = merge $entry (dict "name" $name "pagegate" (printf "%v" $entry.pagegate)) -}} {{ $entries = $entries | append $entry -}} {{ end -}} {{ $entries = sort $entries "weight" -}} {{ range $entries -}} {{ $entry := . -}} {{ $name := .name -}} {{ if not (findRE `^[a-z0-9_-]+$` $name) -}} {{/* Names address assets and partials by path; refuse anything that could traverse outside the plugin namespaces. */ -}} {{ warnidf "docsy-plugin-name" "Ignoring plugin %q: names are restricted to lowercase letters, digits, hyphens, and underscores." $name -}} {{ continue -}} {{ end -}} {{/* A plugin's shim adapts parameters that predate the registry onto its entry; each shim retires with its parameter's deprecation cycle. */ -}} {{ $shim := printf "scripts/plugins/%s-shim.html" $name -}} {{ if templates.Exists (printf "_partials/%s" $shim) -}} {{ $entry = partial $shim (dict "Page" $ "Plugin" $entry) -}} {{ if not (reflect.IsMap $entry) -}} {{ errorf "Plugin shim %s must return the plugin entry (a map); got %T." $shim $entry -}} {{ end -}} {{ end -}} {{ if or (eq $entry.enable false) (eq (lower (printf "%v" $entry.enable)) "false") -}} {{ continue -}} {{ end -}} {{ $gate := printf "%v" $entry.pagegate -}} {{ $defer := and $entry.defer (ne (lower (printf "%v" $entry.defer)) "false") -}} {{ $path := printf "js/plugins/%s.js" $name -}} {{ with resources.Get $path -}} {{ if or (not $gate) ($.Page.Store.Get $gate) -}} {{ with resources.Get (printf "scss/plugins/%s.scss" $name) -}} {{ $css := . | toCSS (dict "transpiler" "dartsass") -}} {{ if hugo.IsProduction -}} {{ $css = $css | minify -}} {{ end -}} {{ $css = $css | fingerprint -}} {{- "\n" -}} {{ end -}} {{ $companion := printf "scripts/plugins/%s.html" $name -}} {{ if templates.Exists (printf "_partials/%s" $companion) -}} {{ partial $companion (dict "Page" $ "Plugin" $entry) -}} {{ end -}} {{ $opts := dict "params" ($entry.options | default dict) -}} {{ if hugo.IsProduction -}} {{ $opts = merge $opts (dict "minify" true) -}} {{ end -}} {{ $js := . | js.Build $opts | fingerprint -}} {{- "\n" -}} {{ end -}} {{ else -}} {{ warnidf "docsy-plugin-missing" "Plugin %q is registered in params.docsy.plugins but assets/%s was not found." $name $path -}} {{ end -}} {{ end -}}
