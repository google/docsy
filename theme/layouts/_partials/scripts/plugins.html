Registry contract, shape guards, and companion conventions: https://www.docsy.dev/project/implementation/script-loading/ */ -}} {{/* The theme declares its plugins under this key (theme/hugo.yaml), so a site's non-map value here, nil included, has already replaced them in Hugo's config merge: say so. */ -}} {{ $registry := dict -}} {{ $docsy := .Site.Params.docsy -}} {{ if not (reflect.IsMap $docsy) -}} {{ warnidf "docsy-plugins-config" "params.docsy must be a map of theme settings (got %v): the theme plugins and their legacy parameters are off. Ignoring it." $name $k -}} {{ end -}} {{ end -}} {{ else if or (eq $value false) (eq (lower (printf "%v" $value)) "false") -}} {{ $entry = merge $entry (dict "enable" false) -}} {{ else -}} {{/* A scalar replaced any theme entry at merge time; the intent is unrecoverable. */ -}} {{ warnidf "docsy-plugin-entry" "Plugin %q: %v is not a plugin entry (a map); write `%s: { enable: true }` to enable it with its defaults. Skipping it." $name $value $name -}} {{ continue -}} {{ end -}} {{/* A gate is a Store flag name; a boolean, nil, or empty value means none. The key is the name; set last so an entry field can't redirect the asset. */ -}} {{ $gate := $entry.pagegate -}} {{ if or (not $gate) (eq (printf "%T" $gate) "bool") }}{{ $gate = "" }}{{ else }}{{ $gate = printf "%v" $gate }}{{ end -}} {{ $options := $entry.options | default dict -}} {{ if not (reflect.IsMap $options) -}} {{ warnidf "docsy-plugin-entry" "Plugin %q: options must be a map (got %v); using none." $name $options -}} {{ $options = dict -}} {{ end -}} {{ $entry = merge $entry (dict "name" $name "pagegate" $gate "options" $options) -}} {{ $entries = $entries | append $entry -}} {{ end -}} {{ $entries = sort $entries "weight" -}} {{ range $entries -}} {{ $entry := . -}} {{ $name := .name -}} {{ if not (findRE `^[a-z0-9_-]+$` $name) -}} {{/* Names address assets and partials by path; refuse anything that could traverse outside the plugin namespaces. */ -}} {{ warnidf "docsy-plugin-name" "Ignoring plugin %q: names are restricted to lowercase letters, digits, hyphens, and underscores." $name -}} {{ continue -}} {{ end -}} {{ if hasSuffix $name "_docsy-shim" -}} {{ warnidf "docsy-plugin-name" "Ignoring plugin %q: the _docsy-shim suffix is reserved for shim partials." $name -}} {{ continue -}} {{ end -}} {{/* A plugin's shim adapts parameters that predate the registry onto its entry; each shim retires with its parameter's deprecation cycle. */ -}} {{ $shim := printf "scripts/plugins/%s_docsy-shim.html" $name -}} {{ if templates.Exists (printf "_partials/%s" $shim) -}} {{ $entry = partial $shim (dict "Page" $ "Plugin" $entry) -}} {{ if not (reflect.IsMap $entry) -}} {{ errorf "Plugin shim %s must return the plugin entry (a map); got %T." $shim $entry -}} {{ end -}} {{ end -}} {{ if or (eq $entry.enable false) (eq (lower (printf "%v" $entry.enable)) "false") -}} {{ continue -}} {{ end -}} {{ $gate := printf "%v" $entry.pagegate -}} {{ $defer := and $entry.defer (ne (lower (printf "%v" $entry.defer)) "false") -}} {{ $path := printf "js/plugins/%s.js" $name -}} {{ with resources.Get $path -}} {{ if or (not $gate) ($.Page.Store.Get $gate) -}} {{ with resources.Get (printf "scss/plugins/%s.scss" $name) -}} {{ $css := . | toCSS (dict "transpiler" "dartsass") -}} {{ if hugo.IsProduction -}} {{ $css = $css | minify -}} {{ end -}} {{ $css = $css | fingerprint -}} {{- "\n" -}} {{ end -}} {{ $companion := printf "scripts/plugins/%s.html" $name -}} {{ if templates.Exists (printf "_partials/%s" $companion) -}} {{ partial $companion (dict "Page" $ "Plugin" $entry) -}} {{ end -}} {{ $opts := dict "params" $entry.options -}} {{ if hugo.IsProduction -}} {{ $opts = merge $opts (dict "minify" true) -}} {{ end -}} {{ $js := . | js.Build $opts | fingerprint -}} {{- "\n" -}} {{ end -}} {{ else -}} {{ warnidf "docsy-plugin-missing" "Plugin %q is registered in params.docsy.plugins but assets/%s was not found." $name $path -}} {{ end -}} {{ end -}}
